cmake_minimum_required(VERSION 3.8)
project(apriltag_detector)

# 1. Set C++ Standard (Required for ROS 2 and Eigen)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()

if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

# find dependencies
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(extender_msgs REQUIRED)
find_package(OpenCV REQUIRED)
find_package(apriltag REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)

############ AprilTag Detector Component ############

include_directories(
  include
)

add_executable(apriltag_detector src/detector.cpp)
add_executable(apriltag_bridge src/bridge.cpp)

ament_target_dependencies(apriltag_detector
  rclcpp
  rclcpp_components
  sensor_msgs
  extender_msgs
  cv_bridge
  apriltag
  tf2_ros
)

target_include_directories(apriltag_detector PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>
)

install(DIRECTORY include/ DESTINATION include)

install(TARGETS
  apriltag_detector
  DESTINATION lib/${PROJECT_NAME}
  ARCHIVE DESTINATION lib
  LIBRARY DESTINATION lib
  RUNTIME DESTINATION bin)

install(DIRECTORY config launch
  DESTINATION share/${PROJECT_NAME}
)

ament_package()
